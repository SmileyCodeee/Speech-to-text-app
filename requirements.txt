# ---------------------------------------------------------------------------
# Speech-to-Text Note Taking Application — requirements.txt
# CLOUD-FRIENDLY VERSION — for Streamlit Community Cloud (free tier, ~1GB RAM)
#
# Deliberately EXCLUDES `transformers` + `torch` (abstractive summarization).
# Those two packages alone add well over 1GB of install size and the BART
# model itself is ~1.6GB in memory — that combination reliably exceeds
# Community Cloud's free-tier memory limit and crashes the app on model
# load, with no clear error in the logs (it just gets silently killed).
#
# The app detects at runtime whether `transformers` is installed and simply
# hides the "Abstractive" summary option if it isn't — so this lighter
# requirements.txt is safe to deploy as-is; nothing will crash, abstractive
# mode just won't be offered.
#
# If you want abstractive summarization too and are deploying somewhere
# with more memory (your own server, Streamlit Cloud paid tier, etc.), use
# requirements-full.txt instead.
# ---------------------------------------------------------------------------

# Core UI
streamlit==1.60.0

# Speech recognition (offline) — faster-whisper (CTranslate2 reimplementation
# of OpenAI Whisper): same model weights/accuracy as openai-whisper, but
# 2-4x faster on CPU via int8 quantized inference. Pulls in ctranslate2
# automatically. Stick to "tiny" or "base" model sizes on Community Cloud —
# "small" and above risk exceeding the free-tier memory limit.
faster-whisper==1.2.1

# Audio decoding/normalization — re-encodes and safely decodes audio before
# it reaches the ASR engine, avoiding silent 0-sample decode failures on
# some browser-recorded files.
soundfile==0.14.0
numpy==2.5.1
av==18.0.0

# Portable ffmpeg fallback — used automatically if ffmpeg isn't found on
# the system PATH (Community Cloud doesn't have it system-wide by default).
imageio-ffmpeg==0.6.0

# Online transcription mode (Google's free Web Speech API) — lightweight,
# no local model, works well within Cloud's memory limits.
SpeechRecognition==3.17.0

# Text processing / keyword extraction (lightweight, multilingual)
yake==0.7.3

# Export formats
python-docx==1.2.0
reportlab==5.0.0
