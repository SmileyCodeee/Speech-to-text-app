# ──────────────────────────────────────────────────────────
# Speech-to-Text Note Taking Application — requirements.txt
# ──────────────────────────────────────────────────────────
# CPU-only throughout — designed for low-spec local machines and
# GPU-less hosts like Render. faster-whisper runs int8-quantized on CPU
# (via ctranslate2, installed automatically as its dependency — no GPU
# drivers or CUDA toolkit required anywhere in this app).
#
# Hybrid architecture:
#   - ASR / structuring / keywords / extractive summary: OFFLINE
#   - Abstractive summary: ONLINE (first run, cached after)
#   - gTTS read aloud: ONLINE
#   - Export: OFFLINE
#
# Only DIRECT dependencies with loose version pins. Transitive deps are
# resolved automatically by pip.
# ──────────────────────────────────────────────────────────

# ── Web UI ──
streamlit>=1.50.0

# ── ASR — faster-whisper (offline, local, CPU-only) ──
faster-whisper>=1.1.0

# ── Audio processing ──
soundfile>=0.12.1
numpy>=1.24.0
imageio-ffmpeg>=0.5.0          # portable ffmpeg binary (fallback if system ffmpeg missing)

# ── Keyword extraction (offline, local) ──
yake>=0.4.5

# ── Export (offline, local) ──
python-docx>=1.1.0
reportlab>=4.2.0

# ── Abstractive summarization (online first run, cached locally after) ──
# CPU-ONLY torch. Without this, `pip install torch` on Linux pulls the
# CUDA build, which drags in ~1.5GB of unused nvidia-* packages
# (nvidia-cublas, nvidia-cudnn, nvidia-cusolver, triton, etc.) — a real
# problem on memory/disk-constrained hosts like Render and low-spec local
# machines, since nothing in this app uses a GPU. This index hosts
# CPU-only wheels for both Windows and Linux.
--extra-index-url https://download.pytorch.org/whl/cpu
transformers>=4.30.0
torch>=2.0.0

# ── Read Notes Aloud (online — gTTS) ──
gTTS>=2.3.0
